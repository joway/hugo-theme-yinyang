{{- /* Render an image. Local images under assets/images are converted to WebP and emitted with a srcset of 800/1200/1600px widths. Anything else is passed through. Params (dict): src (required), alt, title, class, sizes, link (bool: wrap in to the largest variant), eager (bool: loading=eager + fetchpriority=high, for the LCP image) */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $title := .title | default "" -}} {{- $class := .class | default "" -}} {{- $sizes := .sizes | default "(max-width: 800px) 100vw, 800px" -}} {{- $link := .link | default false -}} {{- $eager := .eager | default false -}} {{- $loading := cond $eager `loading="eager" fetchpriority="high"` `loading="lazy"` | safeHTMLAttr -}} {{- /* normalize to an assets path like "images/foo/bar.jpg" */ -}} {{- $path := $src -}} {{- $path = index (split $path "?") 0 -}} {{- $path = strings.TrimPrefix "https://ik.imagekit.io/elsetech/blog/" $path -}} {{- $path = replaceRE `^(\.\./)+` "" $path -}} {{- $path = strings.TrimPrefix "/" $path -}} {{- $img := "" -}} {{- if hasPrefix $path "images/" -}} {{- $img = resources.Get $path -}} {{- end -}} {{- if and $img (in (slice "image/jpeg" "image/png" "image/webp") $img.MediaType.Type) -}} {{- $opts := "webp q85" -}} {{- $widths := slice 800 1200 1600 -}} {{- $variants := slice -}} {{- range $widths -}} {{- if lt . $img.Width -}} {{- $variants = $variants | append ($img.Resize (printf "%dx %s" . $opts)) -}} {{- end -}} {{- end -}} {{- /* the largest variant is the original width if it is under 1600px, else 1600px */ -}} {{- if le $img.Width 1600 -}} {{- $variants = $variants | append ($img.Process $opts) -}} {{- end -}} {{- $main := index $variants (sub (len $variants) 1) -}} {{- $srcset := slice -}} {{- range $variants }}{{ $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) }}{{ end -}} {{ if $link }}{{ end -}} {{- if $link }}{{ end }} {{- else if $img -}} {{- else -}} {{- if hasPrefix $path "images/" -}}{{ warnf "image not found in assets: %s (from %s)" $path $src }}{{- end -}} {{- end -}}
